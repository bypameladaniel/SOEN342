
//For a given connection, a client may only have a single reservation under their name

context ReservationDB
inv : self.reservation -> 
	forAll ( r1, r2: Reservation | r1 <> r2 implies
		r1.booking.trip <> r2.booking.trip or
		r1.client <> r2.client )

//Each reservation is documented by a ticket

context Reservation 
inv : self.ticket <> null

//In a booking each reservation is under a different name

context Reservation
inv: self.booking.reservations -> 
	forALL (r1, r2: Reservation | r1<>r2 implies 
		r1.client.lastname <> r2.client.lastname or
		r1.client.firstname <> r2.client.firstname
